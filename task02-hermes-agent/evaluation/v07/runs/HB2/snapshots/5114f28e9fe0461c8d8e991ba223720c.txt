    function relatedGameForContent(data, item) {

      if (!title || !item.platformId) return null;

      return integratedDemo.visibleFeaturedGames(data).find(game => {
        if (game.platformId !== item.platformId) return false;
        const gameName = String(game.name || '').replace(/[^\p{L}\p{N}]/gu, '');
        if (!gameName) return false;
        if (title.includes(gameName) || gameName.includes(title)) return true;

        let sharedPrefixLength = 0;
        while (
          sharedPrefixLength < title.length &&
          sharedPrefixLength < gameName.length &&
          title[sharedPrefixLength] === gameName[sharedPrefixLength]
        ) {
          sharedPrefixLength += 1;
        }
        return sharedPrefixLength >= 2;

    }

    function renderRankingSchema(data) {
      const schemaScript = document.getElementById('ranking-schema-json');

      const rows = integratedDemo.rankingRows(data);
      const modified = (
        data.sectionUpdatedAt && data.sectionUpdatedAt.homepage
      ) || data.updatedAt || new Date().toISOString().slice(0, 10);
      const itemListElement = rows.map((platform, idx) => {

          "@type": "ListItem",
          "position": idx + 1,
          "item": {
            "@type": "Thing",

            "url": `${window.location.origin}/platforms/${platform.id}`,
            "description": `公開資料完整度 ${platform.score} / 100；此分數不代表平台品質或獲利機率。`
          }
        };
      });
      const schema = {
        "@context": "https://schema.org",
        "@type": "ItemList",
        "name": "休閒娛樂城活動情報站綜合排行榜",
        "description": "依公開資訊完整度、活動更新、遊戲資料、榜單可查性與更新可信度整理的平台資料分排行榜。",
        "dateModified": modified,

    async function renderIntegratedData() {
      if (!integratedDemo) return;
      const data = await geoDataClient.loadFrontendData();

      const updated = document.getElementById('integrated-updated');

      const publicSummary = data.publicSummary || {};
      if (updated) updated.textContent = data.updatedAt;
      if (platformCount) platformCount.textContent =
        publicSummary.observedPlatformCount || data.platforms.length;
      if (jackpotCount) jackpotCount.textContent = integratedDemo.visibleJackpots(data).length;
      const observedCount = document.getElementById('observed-platform-count');
      const platformPageCount = document.getElementById('platform-page-count');
      const rankedPlatformCount = document.getElementById('ranked-platform-count');
      if (observedCount) observedCount.textContent =
        publicSummary.observedPlatformCount || data.platforms.length;
      if (platformPageCount) platformPageCount.textContent =
        publicSummary.platformPageCount || 7;
      if (rankedPlatformCount) rankedPlatformCount.textContent =

      renderIntegratedRanking(data);

      renderRankingSchema(data);
    }

    const searchInput = document.getElementById('siteSearch');
    const filterButtons = [...document.querySelectorAll('.filter-chip')];
    const resultNote = document.getElementById('resultNote');
    let activeFilter = 'all';

    function normalize(value) {
      return String(value || '').trim().toLowerCase();
    }

    function itemText(item) {
      return normalize([
        item.textContent,
        item.dataset.keywords,
        item.dataset.topic
      ].join(' '));
    }

    function matches(item, query) {
      const topics = normalize(item.dataset.topic).split(/\s+/);
      const byFilter = activeFilter === 'all' || topics.includes(activeFilter);
      const bySearch = !query || itemText(item).includes(query);
      return byFilter && bySearch;
    }

    function updateGroup(selector, emptyId, query) {
      const items = [...document.querySelectorAll(selector)];
      let visible = 0;
      items.forEach(item => {
        const ok = matches(item, query);
        item.hidden = !ok;
        if (ok) visible += 1;
      });
      const empty = document.getElementById(emptyId);
      if (empty) empty.classList.toggle('show', visible === 0);
      return visible;
    }

    function applyFilters() {
      const query = normalize(searchInput.value);
      const contentVisible = updateGroup('#events .searchable', 'contentEmpty', query);
      const jackpotVisible = updateGroup('#jackpot .searchable', 'jackpotEmpty', query);
      const gameVisible = updateGroup('#featured-games .searchable', null, query);
      const platformVisible = updateGroup('#platforms .searchable', 'platformEmpty', query);
      const faqVisible = updateGroup('#faq .searchable', 'faqEmpty', query);
      const guideVisible = updateGroup('#selection-guide .searchable', null, query);
      const activityVisible = updateGroup('#activity-compare .searchable', null, query);
      const safetyVisible = updateGroup('#safety-check .searchable', null, query);
      const total = contentVisible + jackpotVisible + gameVisible + platformVisible + faqVisible + guideVisible + activityVisible + safetyVisible;
      const filterLabel = filterButtons.find(button => button.dataset.filter === activeFilter)?.textContent.trim() || '全部';
      resultNote.textContent = query
        ? `搜尋「${searchInput.value.trim()}」｜${filterLabel}｜顯示 ${total} 筆`
        : `${filterLabel}｜顯示 ${total} 筆`;
    }

    filterButtons.forEach(button => {
      button.addEventListener('click', () => {
        activeFilter = button.dataset.filter || 'all';
        filterButtons.forEach(item => item.classList.toggle('active', item === button));
        applyFilters();
      });
    });

    searchInput.addEventListener('input', applyFilters);
    searchInput.addEventListener('change', applyFilters);
    searchInput.addEventListener('search', applyFilters);
    searchInput.addEventListener('keyup', applyFilters);

    document.querySelectorAll('a[href^="#activity-"]').forEach(link => {
      link.addEventListener('click', event => {
        const target = document.querySelector(link.getAttribute('href'));
        if (!(target instanceof HTMLDetailsElement)) return;
        event.preventDefault();
        target.open = true;
        target.scrollIntoView({ behavior: 'smooth', block: 'start' });
        history.replaceState(null, '', link.getAttribute('href'));
      });
    });

    if (location.hash && location.hash.startsWith('#')) {
      try {
        const initialActivity = document.querySelector(location.hash);
        if (initialActivity instanceof HTMLDetailsElement) {
          initialActivity.open = true;
        }
      } catch (e) {
        console.error('Invalid hash selector:', e);
      }
    }

    const toggleBtn = document.getElementById('toggle-expired-btn');
    if (toggleBtn) {
      toggleBtn.addEventListener('click', () => {
        showExpiredGames = !showExpiredGames;
        renderIntegratedFeaturedGames(currentFrontendData);
      });
    }

    let frontendRefreshPromise = null;
    function refreshFrontendData() {
      if (frontendRefreshPromise) return frontendRefreshPromise;
      frontendRefreshPromise = renderIntegratedData()
        .then(applyFilters)
        .finally(() => {
          frontendRefreshPromise = null;
        });
      return frontendRefreshPromise;
    }

    refreshFrontendData();
    window.addEventListener('storage', refreshFrontendData);
    window.addEventListener('focus', refreshFrontendData);
    window.addEventListener('pageshow', refreshFrontendData);
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'visible') refreshFrontendData();
    });

    applyFilters();
  </script>
