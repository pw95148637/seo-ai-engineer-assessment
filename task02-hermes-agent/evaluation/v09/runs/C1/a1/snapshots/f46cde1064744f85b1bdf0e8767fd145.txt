/* 網站點擊統計：只送一個 beacon，不影響右鍵／開新視窗／SEO
 ★ v2 點擊來源（更正）：歸**組**（二級分類）——
 分類頁組塊／首頁分類卡帶 data-src="grp:<gid>" ⇒ 總點擊+1 且 該組+1（分類數＝Σ組同步）；
 熱門／歷史／搜尋等非分類入口只 +總點擊。
 歷史紀錄等只有網址的入口走 hitU()（/?nav_click_url=…，寫回時反查 nav_site id）。 */
(function(){
  function beacon(u){
    try{
      if(navigator.sendBeacon){ navigator.sendBeacon(u); }
      else { var i=new Image(); i.src=u; }
    }catch(e){}
  }
  function hit(t,s){
    var u="/?nav_click="+t;
    if(s){ u+="&src="+encodeURIComponent(s); }
    beacon(u);
  }
  function hitU(url,s){
    var u="/?nav_click_url="+encodeURIComponent(url);
    if(s){ u+="&src="+encodeURIComponent(s); }
    beacon(u);
  }
  window.navHit=hit; window.navHitU=hitU;   /* nav.js 的「歷史紀錄」點擊用（那邊只有網址） */
  document.addEventListener("click",function(ev){
    var a=ev.target;
    while(a && a.tagName!=="A"){ a=a.parentNode; }
    if(!a || !a.getAttribute){ return; }
    var t=a.getAttribute("data-nid");
    var s=a.getAttribute("data-src")||"";
    if(t){ hit(t,s); }
    else if(s){ var h=a.getAttribute("href")||""; if(/^https?:\/\//i.test(h)){ hitU(h,s); } }
  },true);
})();
