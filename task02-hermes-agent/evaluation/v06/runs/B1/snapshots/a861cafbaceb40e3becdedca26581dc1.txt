jQuery(document).ready(function($) {
    // 漢堡選單開關 - 使用 JNews 原生 body.jeg_show_menu 機制
    $('.jeg_mobile_toggle').on('click', function(e) {
        e.preventDefault();
        $('body').toggleClass('jeg_show_menu');
    });
    // 關閉按鈕
    $('.jeg_menu_close').on('click', function(e) {
        e.preventDefault();
        $('body').removeClass('jeg_show_menu');
    });
    // 點擊 overlay 關閉
    $('.jeg_bg_overlay').on('click', function() {
        $('body').removeClass('jeg_show_menu');
    });
    // 點擊頁面其他區域關閉
    $(document).on('mouseup', function(e) {
        if ($('body').hasClass('jeg_show_menu') && !$(e.target).parents('.jeg_mobile_wrapper').length && !$(e.target).hasClass('jeg_mobile_toggle') && !$(e.target).parents('.jeg_mobile_toggle').length) {
            $('body').removeClass('jeg_show_menu');
        }
    });
    // 搜尋按鈕
    $('.jeg_search_toggle').on('click', function(e) {
        e.preventDefault();
        var $wrapper = $(this).closest('.jeg_search_wrapper');
        $wrapper.toggleClass('open');
        if ($wrapper.hasClass('open')) {
            $wrapper.find('.jeg_search_input').focus();
        }
    });
    // 子選單展開
    $('.jeg_main_menu .menu-item-has-children > a').on('mouseenter', function() {
        $(this).siblings('.sub-menu').stop(true,true).slideDown(200);
    });
    $('.jeg_main_menu .menu-item-has-children').on('mouseleave', function() {
        $(this).find('.sub-menu').stop(true,true).slideUp(200);
    });
});
